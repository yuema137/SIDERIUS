import torch
import torch.nn as nn
import torch.nn.functional as F
from pydantic import BaseModel, Field, model_validator
from typing import Self


PLUGIN_MODEL_TYPE = "s04a_baseline_classifier"


class S04aBaselineClassifierConfig(BaseModel):
    model_type: str = Field(default="s04a_baseline_classifier", description="Plugin model type key.")
    segmentation_size: int = Field(default=24, ge=1)
    batch_size: int = Field(default=2, ge=1)
    channels: int = Field(default=16, ge=1)


PLUGIN_CONFIG_CLASS = S04aBaselineClassifierConfig


class S04aBaselineClassifier(nn.Module):
    def __init__(self, config: "S04aBaselineClassifierConfig"):
        super().__init__()
        self.emb = nn.Embedding(7, config.channels)
        self.head = nn.Conv1d(config.channels, 7, kernel_size=1)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        # forward contract: input [B, T] int64 → output [B, 7, T] float32
        out = self.emb(x).permute(0, 2, 1)
        return self.head(out)


PLUGIN_MODEL_CLASS = S04aBaselineClassifier
PLUGIN_OUTPUT_TYPE = "classifier"  # [B, 7, T] → 7-class classification
